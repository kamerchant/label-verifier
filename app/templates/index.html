        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();


        function playTone(type) {

            if (audioCtx.state === 'suspended') audioCtx.resume();

            const osc = audioCtx.createOscillator();

            const gain = audioCtx.createGain();

            osc.connect(gain);

            gain.connect(audioCtx.destination);


            if (type === 'PASS') {

                osc.frequency.setValueAtTime(1046.5, audioCtx.currentTime);

                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);

                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);

                osc.start();

                osc.stop(audioCtx.currentTime + 0.15);

            } else {

                osc.type = 'sawtooth';

                osc.frequency.setValueAtTime(440, audioCtx.currentTime);

                osc.frequency.linearRampToValueAtTime(220, audioCtx.currentTime + 0.4);

                gain.gain.setValueAtTime(0.6, audioCtx.currentTime);

                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);

                osc.start();

                osc.stop(audioCtx.currentTime + 0.4);

            }

        }

            const filtered = rawMasterJobs.filter(j => 

                j.job_card_id.toLowerCase().includes(query) ||

                (j.description && j.description.toLowerCase().includes(query)) ||

                (j.status && j.status.toLowerCase().includes(query))

            );


            const tbody = document.getElementById('master-jobs-rows');

            if (filtered.length === 0) {

                tbody.innerHTML = `<tr><td colspan="7" class="p-4 text-center text-slate-500 italic text-sm">No job cards found matching '${query}'</td></tr>`;

                return;

                    ? `<span class="text-slate-500 block font-mono text-xs">Created: ${j.created_at}</span><span class="text-red-400 block font-mono text-xs">Deleted: ${j.deleted_at}</span>`

        async function handleScan(e) {

            e.preventDefault();

            const input = document.getElementById('barcode-input');

            const code = input.value.trim();

            if (!code) return;

            if (!activeJobId) {

                alert("Please select an active Job Card from the search bar first.");

                input.value = "";

                return;

            }


            const body = new FormData();

            body.append("job_card_id", activeJobId);

            body.append("code", code);


            const res = await fetch('/api/verify', { method: 'POST', body });

            const data = await res.json();

            renderResult(data.result, data.message, code);

            input.value = "";

        }


        function renderResult(result, msg, code) {

            const card = document.getElementById('status-card');

            const heading = document.getElementById('status-heading');

            const detail = document.getElementById('status-detail');

            const badge = document.getElementById('status-badge');

            const dot = document.getElementById('status-dot');

            const codeDisplay = document.getElementById('status-last-code');


            codeDisplay.innerText = code;

            codeDisplay.classList.remove('hidden');

            badge.classList.remove('hidden');


            if (result === 'PASS') {

                playTone('PASS');

                card.className = "linear-subcard px-3.5 sm:px-5 py-3 sm:py-4 rounded flex flex-col md:flex-row items-start md:items-center justify-between gap-2 sm:gap-3 state-pass";

                dot.className = "w-2.5 h-2.5 rounded-full bg-[#2ed573] shrink-0";

                badge.className = "px-2.5 py-0.5 sm:py-1 rounded text-[11px] sm:text-xs font-mono font-bold linear-badge-pass";

                badge.innerText = "PASS";

                heading.innerText = "VALIDATED";

                detail.innerText = `· ${msg}`;

            } else if (result === 'DUPLICATE' || result === 'BLOCKED') {

                playTone('FAIL');

                card.className = "linear-subcard px-3.5 sm:px-5 py-3 sm:py-4 rounded flex flex-col md:flex-row items-start md:items-center justify-between gap-2 sm:gap-3 state-fail";

                dot.className = "w-2.5 h-2.5 rounded-full bg-[#ff4757] shrink-0";

                badge.className = "px-2.5 py-0.5 sm:py-1 rounded text-[11px] sm:text-xs font-mono font-bold linear-badge-fail";

                badge.innerText = "STOP RUN";

                heading.innerText = "COLLISION / BLOCKED";

                detail.innerText = `· ${msg}`;

            } else if (result === 'MISMATCH') {

                playTone('FAIL');

                card.className = "linear-subcard px-3.5 sm:px-5 py-3 sm:py-4 rounded flex flex-col md:flex-row items-start md:items-center justify-between gap-2 sm:gap-3 state-fail";

                dot.className = "w-2.5 h-2.5 rounded-full bg-[#ff4757] shrink-0";

                badge.className = "px-2.5 py-0.5 sm:py-1 rounded text-[11px] sm:text-xs font-mono font-bold linear-badge-fail";

                badge.innerText = "MISMATCH";

                heading.innerText = "BATCH CONFLICT";

                detail.innerText = `· ${msg}`;

            } else {

                playTone('FAIL');

                card.className = "linear-subcard px-3.5 sm:px-5 py-3 sm:py-4 rounded flex flex-col md:flex-row items-start md:items-center justify-between gap-2 sm:gap-3 state-warn";

                dot.className = "w-2.5 h-2.5 rounded-full bg-[#ffa502] shrink-0";

                badge.className = "px-2.5 py-0.5 sm:py-1 rounded text-[11px] sm:text-xs font-mono font-bold linear-badge-warn";

                badge.innerText = "UNKNOWN";

                heading.innerText = "UNREGISTERED CODE";

                detail.innerText = `· ${msg}`;

            }


            const now = new Date().toLocaleTimeString();

            recentScans.unshift({ code, result, time: now, user: currentUser.username });

            if (recentScans.length > 10) recentScans.pop();

            renderStreamTable();

        }

            const tbody = document.getElementById('recent-scans-rows');

            tbody.innerHTML = recentScans.map((s, idx) => {

                let badge = "linear-badge-pass";

                if (s.result !== 'PASS') badge = "linear-badge-fail";

                return `

                    <tr class="hover:bg-white/[0.02]">

                `;

        }


        async function uploadJob(e, overrideDeletedWarning = false) {

            if (e) e.preventDefault();

            const btn = document.getElementById('upload-btn');

            btn.disabled = true;

            btn.innerText = "Processing...";


            const body = new FormData();

            body.append("job_card_id", document.getElementById('new-job-id').value.trim());

            body.append("description", document.getElementById('new-job-desc').value.trim());

            body.append("file", document.getElementById('new-job-file').files[0]);

            body.append("override_deleted_warning", overrideDeletedWarning);


            const res = await fetch('/api/jobs/create', { method: 'POST', body });

            const data = await res.json();


            if (res.status === 409 && data.status === "deleted_duplicate_warning") {

                document.getElementById('deleted-dup-warning-msg').innerText = 

                    `Code '${data.code}' exists in deleted Job Card '${data.deleted_job_card}'. Would you like to continue creating this job anyway?`;

                document.getElementById('deleted-dup-warning-modal').classList.remove('hidden');

                btn.disabled = false;

                btn.innerText = "Create Job";

                return;

            }


            if (res.ok) {

                alert(`Job Card ${data.job_card_id} created with ${data.total_extracted.toLocaleString()} serial codes.`);

                toggleModal('upload-modal');

                document.getElementById('deleted-dup-warning-modal').classList.add('hidden');

                document.getElementById('create-job-form').reset();

                selectJob(data.job_card_id, escape(document.getElementById('new-job-desc').value.trim()), 'ACTIVE', 0, data.total_extracted);

            } else {

                alert(data.detail || "Failed to create job");

            }


            btn.disabled = false;

            btn.innerText = "Create Job";

        }

        async function toggleUploadPermission(username, allowed) {

            const body = new FormData();

            body.append("can_upload", allowed);


            const res = await fetch(`/api/users/${username}/toggle-upload`, {

                method: 'POST',

                body: body

            });


            if (res.ok) {

                if (username === currentUser.username) checkSession();

                openUserModal();

            } else {

                alert("Failed to update user authorization.");

            }

        }
