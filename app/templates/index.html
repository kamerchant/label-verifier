    <script>

        let activeJobId = null;

        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();


        // Synthesize shopfloor alerts

        function playTone(type) {

            if (audioCtx.state === 'suspended') audioCtx.resume();

            const osc = audioCtx.createOscillator();

            const gain = audioCtx.createGain();

            osc.connect(gain);

            gain.connect(audioCtx.destination);


            if (type === 'PASS') {

                osc.frequency.setValueAtTime(1046.5, audioCtx.currentTime); // High C

                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);

                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);

                osc.start();

                osc.stop(audioCtx.currentTime + 0.15);

            } else {

                // Harsh industrial warning tone

                osc.type = 'sawtooth';

                osc.frequency.setValueAtTime(440, audioCtx.currentTime);

                osc.frequency.linearRampToValueAtTime(220, audioCtx.currentTime + 0.4);

                gain.gain.setValueAtTime(0.6, audioCtx.currentTime);

                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);

                osc.start();

                osc.stop(audioCtx.currentTime + 0.4);

            }

        }


        async function fetchJobs() {

            const res = await fetch('/api/jobs');

            const jobs = await res.json();

            const list = document.getElementById('job-list');

            list.innerHTML = jobs.map(j => `

                <div onclick="selectJob('${j.id}', '${j.status}', ${j.consumed}, ${j.total})" 

                     class="p-3 bg-slate-700/50 hover:bg-slate-700 border border-slate-600 rounded cursor-pointer ${activeJobId === j.id ? 'border-blue-400 bg-slate-700' : ''}">

                    <div class="flex justify-between items-center">

                        <span class="font-bold font-mono text-sm">${j.id}</span>

                        <span class="text-xs px-2 py-0.5 rounded ${j.status === 'ACTIVE' ? 'bg-green-800' : 'bg-red-800'}">${j.status}</span>

                    </div>

                    <div class="text-xs text-slate-400 mt-1">${j.consumed.toLocaleString()} / ${j.total.toLocaleString()} verified</div>

                </div>

            `).join('');

        }


        function selectJob(id, status, consumed, total) {

            activeJobId = id;

            document.getElementById('active-badge').innerText = id;

            document.getElementById('active-badge').className = "px-3 py-1 bg-green-600 rounded text-xs uppercase font-bold tracking-wide";

            document.getElementById('end-job-btn').disabled = (status !== 'ACTIVE');

            document.getElementById('counter').innerText = `${consumed.toLocaleString()} / ${total.toLocaleString()}`;

            fetchJobs();

            resetInput();

        }


        async function handleScan(e) {

            e.preventDefault();

            const input = document.getElementById('barcode-input');

            const code = input.value.trim();

            if (!code) return;

            if (!activeJobId) {

                alert("Please select a Job Card first.");

                input.value = "";

                return;

            }


            const body = new FormData();

            body.append("job_card_id", activeJobId);

            body.append("code", code);


            const res = await fetch('/api/verify', { method: 'POST', body });

            const data = await res.json();

            renderResult(data.result, data.message, code);

            input.value = "";

            fetchJobs();

        }


        function renderResult(result, msg, code) {

            const card = document.getElementById('status-card');

            const heading = document.getElementById('status-heading');

            const detail = document.getElementById('status-detail');

            const badge = document.getElementById('status-badge');


            card.className = "p-8 rounded-lg border border-slate-700 flex flex-col items-center justify-center min-h-[260px] text-center transition-colors duration-200";


            if (result === 'PASS') {

                playTone('PASS');

                card.classList.add('flash-pass');

                badge.innerText = "PASS";

                heading.innerText = "VERIFIED";

                detail.innerText = `Code: ${code}`;

            } else if (result === 'DUPLICATE' || result === 'BLOCKED') {

                playTone('FAIL');

                card.classList.add('flash-fail');

                badge.innerText = "DUPLICATE / BLOCKED";

                heading.innerText = "STOP PRESS";

                detail.innerText = msg;

            } else if (result === 'MISMATCH') {

                playTone('FAIL');

                card.classList.add('flash-fail');

                badge.innerText = "WRONG JOB";

                heading.innerText = "WRONG BATCH";

                detail.innerText = msg;

            } else {

                playTone('FAIL');

                card.classList.add('flash-warn');

                badge.innerText = "UNKNOWN";

                heading.innerText = "INVALID CODE";

                detail.innerText = msg;

            }

        }


        async function completeJob() {

            if (!activeJobId || !confirm(`Mark ${activeJobId} completed and block remaining unprinted codes?`)) return;

            await fetch(`/api/jobs/${activeJobId}/complete`, { method: 'POST' });

            fetchJobs();

            resetInput();

        }


        function toggleUploadModal() {

            document.getElementById('upload-modal').classList.toggle('hidden');

        }


        async function uploadJob(e) {

            e.preventDefault();

            const btn = document.getElementById('upload-btn');

            btn.disabled = true;

            btn.innerText = "Extracting & Uploading...";


            const body = new FormData();

            body.append("job_card_id", document.getElementById('new-job-id').value.trim());

            body.append("columns", document.getElementById('target-columns').value.trim());

            body.append("file", document.getElementById('new-job-file').files[0]);


            try {

                const res = await fetch('/api/jobs/create', { 

                    method: 'POST', 

                    body: body 

                });


                if (res.ok) {

                    const data = await res.json();

                    alert(`Success! Imported ${data.total_extracted.toLocaleString()} unique codes for Job Card: ${data.job_card_id}`);

                    toggleUploadModal();

                    fetchJobs();

                    e.target.reset();

                } else {

                    const err = await res.json();

                    alert(`Error: ${err.detail || "Upload failed. Please check the file."}`);

                }

            } catch (networkError) {

                alert("Network error: Could not reach the server. Check your connection.");

            } finally {

                btn.disabled = false;

                btn.innerText = "Upload";

            }

        }


        function resetInput() {

            const input = document.getElementById('barcode-input');

            input.focus();

        }


        // Auto-refocus scanner input on accidental shopfloor clicks

        document.addEventListener('click', (e) => {

            if (e.target.tagName !== 'INPUT' && e.target.tagName !== 'BUTTON') {

                resetInput();

            }

        });


        fetchJobs();

    </script>
